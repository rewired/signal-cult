cmake_minimum_required(VERSION 3.24)
set(CMAKE_CUDA_ARCHITECTURES "75-real;86-real;89-real;89-virtual" CACHE STRING "CUDA architectures for BROKEN FM")
project(BrokenFm VERSION 0.1.0 LANGUAGES CXX CUDA)

option(BROKEN_FM_BUILD_OFX "Build the BROKEN FM OpenFX plugin" ON)
option(BROKEN_FM_BUILD_TESTS "Build BROKEN FM native smoke tests" ON)
if(BROKEN_FM_BUILD_TESTS)
  enable_testing()
endif()

add_subdirectory(native)
if(BROKEN_FM_BUILD_OFX)
  add_subdirectory(hosts/ofx)
endif()
if(BROKEN_FM_BUILD_TESTS)
  add_executable(broken_fm_core_smoke native/tests/core_smoke.cpp)
  target_link_libraries(broken_fm_core_smoke PRIVATE BrokenFmCore)
  if(MSVC)
    target_link_options(broken_fm_core_smoke PRIVATE /NODEFAULTLIB:LIBCMT)
  endif()
  add_test(NAME broken_fm_core_smoke COMMAND broken_fm_core_smoke)
endif()
