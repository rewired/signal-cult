cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT CRT_SIM_LEGACY_ENTRY_ACTIVE)
  project(CrtSimNative LANGUAGES CXX)
  enable_testing()
  set(CRT_SIM_LEGACY_BUNDLE ON)
  set(CRT_SIM_LEGACY_ENTRY_ACTIVE ON)
  add_subdirectory(".." "crt-sim")
  return()
endif()
add_library(CrtSimCore STATIC src/cpu.cpp)
target_compile_features(CrtSimCore PUBLIC cxx_std_17)
target_include_directories(CrtSimCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set_target_properties(CrtSimCore PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CRT_ENABLE_CUDA)
  target_sources(CrtSimCore PRIVATE src/cuda.cu)
  set_target_properties(CrtSimCore PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_RUNTIME_LIBRARY Static)
endif()
if(MSVC)
  target_compile_options(CrtSimCore PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
endif()
