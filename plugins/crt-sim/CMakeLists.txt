cmake_minimum_required(VERSION 3.24)
project(CrtSim VERSION 0.7.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CUDA_ARCHITECTURES "75-real;86-real;89-real;89-virtual" CACHE STRING "CUDA architectures")
option(CRT_ENABLE_CUDA "Build CRT CUDA acceleration" ON)
option(CRT_SIM_BUILD_OFX "Build the CRT OpenFX plugin" ON)
option(CRT_SIM_BUILD_TESTS "Build CRT native tests" ON)
if(CRT_ENABLE_CUDA)
  enable_language(CUDA)
endif()
find_program(NODE_EXECUTABLE node REQUIRED)
execute_process(COMMAND "${NODE_EXECUTABLE}" scripts/generate-native.mjs WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${NODE_EXECUTABLE}" tools/generate-parameter-contract.mjs WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory(native)
if(CRT_SIM_BUILD_OFX)
  add_subdirectory(hosts/ofx)
endif()
if(CRT_SIM_BUILD_TESTS)
  enable_testing()
  add_executable(crt_core_tests native/tests/core.cpp)
  target_link_libraries(crt_core_tests PRIVATE CrtSimCore)
  if(CRT_ENABLE_CUDA)
    target_compile_definitions(crt_core_tests PRIVATE CRT_WITH_CUDA)
  endif()
  add_test(NAME crt_core COMMAND crt_core_tests)
  add_executable(crt_color_tests native/tests/color.cpp)
  target_link_libraries(crt_color_tests PRIVATE CrtSimCore)
  target_include_directories(crt_color_tests PRIVATE hosts/ofx/src)
  add_test(NAME crt_color COMMAND crt_color_tests)
  if(CRT_ENABLE_CUDA)
    add_executable(crt_cuda_tests native/tests/cuda.cu)
    target_link_libraries(crt_cuda_tests PRIVATE CrtSimCore)
    set_target_properties(crt_cuda_tests PROPERTIES CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Static)
    add_test(NAME crt_cuda COMMAND crt_cuda_tests)
  endif()
  if(CRT_SIM_BUILD_OFX)
    foreach(kind abi host)
      add_executable(crt_${kind}_tests native/tests/${kind}.cpp)
      target_include_directories(crt_${kind}_tests PRIVATE native/third_party/openfx/include)
      add_test(NAME crt_${kind} COMMAND crt_${kind}_tests "$<TARGET_FILE:CrtSimOFX>")
    endforeach()
  endif()
endif()
