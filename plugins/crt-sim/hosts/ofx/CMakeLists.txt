include(FetchContent)
set(NLOHMANN_JSON_ROOT "" CACHE PATH "Path to a nlohmann/json checkout")
if(NLOHMANN_JSON_ROOT)
  add_library(CrtSimJson INTERFACE)
  target_include_directories(CrtSimJson INTERFACE "${NLOHMANN_JSON_ROOT}/include")
  set(CRT_SIM_JSON_TARGET CrtSimJson)
else()
  if(NOT TARGET nlohmann_json::nlohmann_json)
    FetchContent_Declare(nlohmann_json_source GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.11.3)
    FetchContent_MakeAvailable(nlohmann_json_source)
  endif()
  set(CRT_SIM_JSON_TARGET nlohmann_json::nlohmann_json)
endif()
set(CRT_SIM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(CRT_SIM_BUNDLE "${CMAKE_BINARY_DIR}/bundle/RewiredCRT.ofx.bundle")
if(CRT_SIM_LEGACY_BUNDLE)
  set(CRT_SIM_BUNDLE "${CMAKE_BINARY_DIR}/RewiredCRT.ofx.bundle")
endif()
add_library(CrtSimOFX MODULE src/crt_sim_ofx.cpp)
target_include_directories(CrtSimOFX PRIVATE "${CRT_SIM_ROOT}/native/third_party/openfx/include")
target_link_libraries(CrtSimOFX PRIVATE CrtSimCore ${CRT_SIM_JSON_TARGET} shell32 ole32 advapi32)
target_compile_definitions(CrtSimOFX PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(CRT_ENABLE_CUDA)
  target_compile_definitions(CrtSimOFX PRIVATE CRT_WITH_CUDA)
  target_sources(CrtSimOFX PRIVATE src/preview_cuda.cu)
  set_target_properties(CrtSimOFX PROPERTIES CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Static)
endif()
if(MSVC)
  target_compile_options(CrtSimOFX PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
endif()
set_target_properties(CrtSimOFX PROPERTIES PREFIX "" OUTPUT_NAME "RewiredCRT" SUFFIX ".ofx"
  RUNTIME_OUTPUT_DIRECTORY "${CRT_SIM_BUNDLE}/Contents/Win64" LIBRARY_OUTPUT_DIRECTORY "${CRT_SIM_BUNDLE}/Contents/Win64")
foreach(config RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
  set_target_properties(CrtSimOFX PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CRT_SIM_BUNDLE}/Contents/Win64" LIBRARY_OUTPUT_DIRECTORY_${config} "${CRT_SIM_BUNDLE}/Contents/Win64")
endforeach()
configure_file(resources/Info.plist.in "${CRT_SIM_BUNDLE}/Contents/Info.plist" @ONLY)
configure_file("${CRT_SIM_ROOT}/presets/crt-presets.json" "${CRT_SIM_BUNDLE}/Contents/Resources/presets.json" COPYONLY)
configure_file("${CRT_SIM_ROOT}/native/third_party/openfx/LICENSE.md" "${CRT_SIM_BUNDLE}/Contents/Resources/OpenFX-LICENSE.md" COPYONLY)
foreach(file README.md COLOR_MANAGEMENT.md)
  configure_file("${CRT_SIM_ROOT}/native/${file}" "${CRT_SIM_BUNDLE}/Contents/Resources/${file}" COPYONLY)
endforeach()
if(CRT_SIM_BUILD_TESTS)
  enable_testing()
  add_executable(crt_companion_tests "${CRT_SIM_ROOT}/native/tests/companion.cpp")
  target_include_directories(crt_companion_tests PRIVATE src)
  target_link_libraries(crt_companion_tests PRIVATE CrtSimCore ${CRT_SIM_JSON_TARGET})
  add_test(NAME crt_companion COMMAND crt_companion_tests "${CRT_SIM_ROOT}/presets/crt-presets.json")
endif()

configure_file(resources/JSON-LICENSE.MIT "${CRT_SIM_BUNDLE}/Contents/Resources/JSON-LICENSE.MIT" COPYONLY)
if(CRT_SIM_BUILD_TESTS AND CRT_ENABLE_CUDA)
  add_executable(crt_preview_tests "${CRT_SIM_ROOT}/native/tests/preview.cu" src/preview_cuda.cu)
  target_include_directories(crt_preview_tests PRIVATE src)
  target_link_libraries(crt_preview_tests PRIVATE CrtSimCore)
  target_compile_definitions(crt_preview_tests PRIVATE CRT_WITH_CUDA)
  set_target_properties(crt_preview_tests PROPERTIES CUDA_STANDARD 17 CUDA_RUNTIME_LIBRARY Static)
  add_test(NAME crt_preview COMMAND crt_preview_tests)
endif()
