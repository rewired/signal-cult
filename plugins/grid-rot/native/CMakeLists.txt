add_library(GridRotCore STATIC src/core.cpp)
target_compile_features(GridRotCore PUBLIC cxx_std_17)
target_include_directories(GridRotCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set_target_properties(GridRotCore PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(GRID_ROT_ENABLE_CUDA)
  target_sources(GridRotCore PRIVATE src/core_cuda.cu)
  target_compile_definitions(GridRotCore PUBLIC GRID_ROT_WITH_CUDA)
  set_target_properties(GridRotCore PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_RUNTIME_LIBRARY Static)
  target_compile_options(GridRotCore PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
endif()
if(MSVC)
  target_compile_options(GridRotCore PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
endif()
