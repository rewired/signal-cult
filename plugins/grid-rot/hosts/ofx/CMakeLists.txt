set(GRID_ROT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(GRID_ROT_BUNDLE "${CMAKE_BINARY_DIR}/bundle/GridRot.ofx.bundle")
set(OPENFX_ROOT "" CACHE PATH "Path to an OpenFX SDK checkout")
if(OPENFX_ROOT)
  set(GRID_ROT_OPENFX_INCLUDE "${OPENFX_ROOT}/include")
else()
  set(GRID_ROT_OPENFX_INCLUDE "${GRID_ROT_ROOT}/../crt-sim/native/third_party/openfx/include")
endif()
if(NOT EXISTS "${GRID_ROT_OPENFX_INCLUDE}/ofxImageEffect.h")
  message(FATAL_ERROR "OpenFX headers not found. Set OPENFX_ROOT or keep the bundled CRT SIM SDK available.")
endif()
add_library(GridRotOFX MODULE src/grid_rot_ofx.cpp)
target_include_directories(GridRotOFX PRIVATE "${GRID_ROT_OPENFX_INCLUDE}")
target_link_libraries(GridRotOFX PRIVATE GridRotCore)
target_compile_definitions(GridRotOFX PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(GridRotOFX PROPERTIES PREFIX "" OUTPUT_NAME "GridRot" SUFFIX ".ofx" RUNTIME_OUTPUT_DIRECTORY "${GRID_ROT_BUNDLE}/Contents/Win64" LIBRARY_OUTPUT_DIRECTORY "${GRID_ROT_BUNDLE}/Contents/Win64")
foreach(config RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
 set_target_properties(GridRotOFX PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${GRID_ROT_BUNDLE}/Contents/Win64" LIBRARY_OUTPUT_DIRECTORY_${config} "${GRID_ROT_BUNDLE}/Contents/Win64")
endforeach()
configure_file(resources/Info.plist.in "${GRID_ROT_BUNDLE}/Contents/Info.plist" @ONLY)
configure_file("${GRID_ROT_ROOT}/native/README.md" "${GRID_ROT_BUNDLE}/Contents/Resources/README.md" COPYONLY)
