add_library(RasterRuptureCore STATIC src/core.cpp)
target_compile_features(RasterRuptureCore PUBLIC cxx_std_17)
target_include_directories(RasterRuptureCore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/generated")
set_target_properties(RasterRuptureCore PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(RASTER_RUPTURE_ENABLE_CUDA)
  target_sources(RasterRuptureCore PRIVATE src/core_cuda.cu)
  target_compile_definitions(RasterRuptureCore PUBLIC RASTER_RUPTURE_WITH_CUDA)
  set_target_properties(RasterRuptureCore PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_RUNTIME_LIBRARY Static)
  target_compile_options(RasterRuptureCore PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
endif()
if(MSVC)
  target_compile_options(RasterRuptureCore PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
endif()
