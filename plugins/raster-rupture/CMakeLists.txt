cmake_minimum_required(VERSION 3.24)
project(RasterRupture VERSION 0.2.0 LANGUAGES CXX)
option(RASTER_RUPTURE_ENABLE_CUDA "Build CUDA acceleration" ON)
option(RASTER_RUPTURE_BUILD_OFX "Build the OpenFX plugin" ON)
option(RASTER_RUPTURE_BUILD_TESTS "Build native tests" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(RASTER_RUPTURE_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES "75-real;86-real;89-real;89-virtual" CACHE STRING "CUDA architectures")
  endif()
  enable_language(CUDA)
endif()
find_program(NODE_EXECUTABLE node REQUIRED)
execute_process(COMMAND "${NODE_EXECUTABLE}" tools/generate-native-contract.mjs WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory(native)
if(RASTER_RUPTURE_BUILD_OFX)
  add_subdirectory(hosts/ofx)
endif()
if(RASTER_RUPTURE_BUILD_TESTS)
  enable_testing()
  add_executable(raster_rupture_core_tests native/tests/core_smoke.cpp)
  target_link_libraries(raster_rupture_core_tests PRIVATE RasterRuptureCore)
  add_test(NAME raster_rupture_core COMMAND raster_rupture_core_tests)
endif()
