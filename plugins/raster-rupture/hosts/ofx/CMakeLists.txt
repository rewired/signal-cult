set(RASTER_RUPTURE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(RASTER_RUPTURE_BUNDLE "${CMAKE_BINARY_DIR}/bundle/RasterRupture.ofx.bundle")
set(OPENFX_ROOT "" CACHE PATH "Path to an OpenFX SDK checkout")
if(OPENFX_ROOT)
  set(RASTER_RUPTURE_OPENFX_INCLUDE "${OPENFX_ROOT}/include")
else()
  set(RASTER_RUPTURE_OPENFX_INCLUDE "${CMAKE_SOURCE_DIR}/plugins/crt-sim/native/third_party/openfx/include")
endif()
if(NOT EXISTS "${RASTER_RUPTURE_OPENFX_INCLUDE}/ofxImageEffect.h")
  message(FATAL_ERROR "OpenFX headers not found. Set OPENFX_ROOT or keep the bundled CRT SIM SDK available.")
endif()
add_library(RasterRuptureOFX MODULE src/raster_rupture_ofx.cpp)
target_include_directories(RasterRuptureOFX PRIVATE "${RASTER_RUPTURE_OPENFX_INCLUDE}")
target_link_libraries(RasterRuptureOFX PRIVATE RasterRuptureCore)
target_compile_definitions(RasterRuptureOFX PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(RasterRuptureOFX PROPERTIES PREFIX "" OUTPUT_NAME "RasterRupture" SUFFIX ".ofx"
  RUNTIME_OUTPUT_DIRECTORY "${RASTER_RUPTURE_BUNDLE}/Contents/Win64" LIBRARY_OUTPUT_DIRECTORY "${RASTER_RUPTURE_BUNDLE}/Contents/Win64")
foreach(config RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
  set_target_properties(RasterRuptureOFX PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${RASTER_RUPTURE_BUNDLE}/Contents/Win64" LIBRARY_OUTPUT_DIRECTORY_${config} "${RASTER_RUPTURE_BUNDLE}/Contents/Win64")
endforeach()
configure_file(resources/Info.plist.in "${RASTER_RUPTURE_BUNDLE}/Contents/Info.plist" @ONLY)
configure_file("${RASTER_RUPTURE_ROOT}/native/README.md" "${RASTER_RUPTURE_BUNDLE}/Contents/Resources/README.md" COPYONLY)
